// Allow sitectl to provide multiple trusted host patterns.
if (isset($libops_env) && is_callable($libops_env)) {
  $trusted_host_patterns_raw = $libops_env('DRUPAL_TRUSTED_HOST_PATTERNS');
}
if (!empty($trusted_host_patterns_raw)) {
  $trusted_host_patterns = preg_split('/[\r\n,]+/', $trusted_host_patterns_raw, -1, PREG_SPLIT_NO_EMPTY);
  $trusted_host_patterns = array_values(array_filter(array_map('trim', $trusted_host_patterns), 'strlen'));
  if (!empty($trusted_host_patterns)) {
    $settings['trusted_host_patterns'] = $trusted_host_patterns;
  }
}
